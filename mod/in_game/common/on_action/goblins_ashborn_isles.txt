monthly_country_pulse = { on_actions = { ga_ashborn_intro_monthly cm_demo_monthly ga_exploration_monthly ga_economy_monthly } }
# Run before Cindermaw's separate starting-unit grant.
ga_ashborn_intro_monthly = {
    trigger = {
        ga_is_goblin = yes
        NOT = { has_variable = ga_ashborn_intro_seen }
        # Older saves dispatched Cindermaw's introduction with this flag.
        NOT = { AND = { tag = CDM has_variable = cm_demo_initialized } }
    }
    effect = {
        set_variable = { name = ga_ashborn_intro_seen value = yes }
        trigger_event_non_silently = { id = cindermaw.1 }
    }
}
cm_demo_monthly = {
    trigger = { tag = CDM }
    effect = {
        if = {
            limit = {
                NOT = { has_variable = cm_demo_initialized }
                owns = location:cm_hooktooth
            }
            set_variable = { name = cm_demo_initialized value = yes }
            location:cm_hooktooth = {
                create_sub_unit = unit_type:a_footmen
                create_sub_unit = unit_type:n_traditional_galley
                create_sub_unit = unit_type:n_traditional_galley
                create_sub_unit = unit_type:n_cog
                create_sub_unit = unit_type:n_cog
                create_sub_unit = unit_type:n_cog
            }
        }
    }
}
