goblins_exploration.1 = {
    type = country_event
    outcome = neutral
    title = goblins_exploration.1.title
    desc = goblins_exploration.1.desc
    trigger = { OR = { tag = CDM tag = QBR tag = RHK tag = SFK tag = SWK } }
    image = "gfx/interface/illustrations/event/ashborn/departure.dds"
    option = {
        name = goblins_exploration.1.east
        trigger = { gold >= 5 NOT = { has_variable = ga_charted_east } }
        add_gold = -5
        trigger_event_non_silently = { id = goblins_exploration.2 months = 4 }
        ai_chance = { factor = 10 }
    }
    option = {
        name = goblins_exploration.1.wait
        remove_variable = ga_exploration_pending
        set_variable = { name = ga_exploration_cooldown months = 6 }
        ai_chance = { factor = 1 }
    }
}
